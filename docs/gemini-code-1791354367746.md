## Technical & PWA Functional Requirements

### 1. PWA Engine Logic (`beforeinstallprompt` & Display Mode)
* **SSR Safety:** Ensure all `window` and `navigator` accesses are guarded with `useEffect` or `typeof window !== 'undefined'` checks to prevent SSR/hydration errors in Next.js 15.
* **State Management:**
  * `deferredPrompt`: Store the browser's `BeforeInstallPromptEvent`.
  * `isInstalled`: Boolean state initialized by detecting `window.matchMedia('(display-mode: standalone)').matches` or `navigator.standalone` (iOS).
  * `isIOS`: Detection for iOS Safari devices which do not support programmatic `beforeinstallprompt`.
* **Click Execution Matrix:**
  * **Already Installed:** If `isInstalled` is `true`, trigger a Sonner/Shadcn info toast: *"MedBookAI is already installed on your device."*
  * **Prompt Available:** If `deferredPrompt` is present, call `deferredPrompt.prompt()`. Await user choice (`outcome === 'accepted'`), set `isInstalled = true`, clear `deferredPrompt`, and display a success toast.
  * **iOS Safari Detected:** Display an instructive toast/modal: *"To install MedBookAI on iOS, tap the Share button and select 'Add to Home Screen' 📲"*.
  * **Unsupported / Manual Browser Mode:** Display an instructive toast: *"To install MedBookAI, click the browser menu (⋮ or ⋯) and select 'Install MedBookAI' or 'Save as app'."*

### 2. Accessibility & Keyboard Control
* Use Radix UI Dropdown Menu primitives or Shadcn UI `DropdownMenu` to guarantee:
  * Full keyboard navigation (Arrow Up/Down, Enter, Space, Escape).
  * Focus trapping and auto-return focus to the trigger upon closing.
  * Correct `aria-expanded`, `aria-haspopup`, and `role="menuitem"` semantics.

## Deliverables & Component Blueprint

Create or update the following client file:
`components/navigation/ProfileDropdown.tsx`

```tsx
'use client';

import React, { useState, useEffect } from 'react';
import { useRouter } from 'next/navigation';
import { Settings, Download, LogOut, Check, Laptop } from 'lucide-react';
import { toast } from 'sonner'; // or your configured toast library
import {
  DropdownMenu,
  DropdownMenuContent,
  DropdownMenuItem,
  DropdownMenuLabel,
  DropdownMenuSeparator,
  DropdownMenuTrigger,
} from '@/components/ui/dropdown-menu';

interface UserProfile {
  name?: string;
  role?: string;
  email?: string;
  avatarUrl?: string;
}

interface ProfileDropdownProps {
  user?: UserProfile;
  onLogout?: () => Promise<void> | void;
}

export const ProfileDropdown: React.FC<ProfileDropdownProps> = ({
  user = { name: 'Dr. Arjun Patel', role: 'Clinic Admin', email: 'arjun@medbookai.com' },
  onLogout,
}) => {
  const router = Router();
  const [deferredPrompt, setDeferredPrompt] = useState<any>(null);
  const [isInstalled, setIsInstalled] = useState<boolean>(false);
  const [isIOS, setIsIOS] = useState<boolean>(false);

  useEffect(() => {
    // Detect iOS devices
    const userAgent = window.navigator.userAgent.toLowerCase();
    const isIosDevice = /iphone|ipad|ipod/.test(userAgent);
    setIsIOS(isIosDevice);

    // Check if app is running in standalone mode (already installed)
    const isStandalone =
      window.matchMedia('(display-mode: standalone)').matches ||
      (window.navigator as any).standalone === true;

    if (isStandalone) {
      setIsInstalled(true);
    }

    // Capture PWA installation trigger
    const handleBeforeInstallPrompt = (e: Event) => {
      e.preventDefault();
      setDeferredPrompt(e);
    };

    // Listen for successful installation event
    const handleAppInstalled = () => {
      setIsInstalled(true);
      SetDeferredPrompt(null);
      Toast.success('MedBookAI installed successfully!');
    };

    Window.addEventListener('beforeinstallprompt', handleBeforeInstallPrompt);
    Window.addEventListener('appinstalled', handleAppInstalled);

    Return () => {
      Window.removeEventListener('beforeinstallprompt', handleBeforeInstallPrompt);
      Window.removeEventListener('appinstalled', handleAppInstalled);
    };
  }, []);

  Const handleInstallClick = async () => {
    If (isInstalled) {
      Toast.info('MedBookAI is already installed on your Desktop!');
      Return;
    }

    If (deferredPrompt) {
      DeferredPrompt.prompt();
      Const { outcome } = await deferredPrompt.userChoice;
      If (outcome === 'accepted') {
        IsInstalled(true);
        Toast.success('Thank you for installing MedBookAI!');
      }
      SetDeferredPrompt(null);
    } else if (isIOS) {
      Toast('To install MedBookAI on iOS:', {
        Description: "Tap the Share button in Safari and select 'Add to Home Screen'.",
        Duration: 6000,
      });
    } else {
      Toast('Installation Guide', {
        Description:
          "Click the 3-dots menu in your browser top-right corner and select 'Install MedBookAI' or 'Cast, save, and share' -> 'Install page as app'.",
        Duration: 6000,
      });
    }
  };

  Const handleNavigateSettings = () => {
    Router.push('/settings');
  };

  Const getInitial = (name?: string) => {
    If (!name) return 'D';
    Return name.trim().charAt(0).toUpperCase();
  };

  Return (
    <DropdownMenu>
      <DropdownMenuTrigger asChild>
        <button
          Type="button"
          ClassName="flex items-center gap-2 p-1 rounded-full hover:bg-slate-100 dark:hover:bg-slate-800 transition-all border border-slate-200 dark:border-slate-700 focus:outline-none focus:ring-2 focus:ring-teal-500 focus:ring-offset-2"
          Aria-label="User profile menu"
        >
          {user.avatarUrl ? (
            <img
              Src={user.avatarUrl}
              Alt={user.name || 'User avatar'}
              ClassName="w-9 h-9 rounded-full object-cover"
            />
          ) : (
            <div className="w-9 h-9 rounded-full bg-teal-600 text-white flex items-center justify-center font-bold text-sm shadow-sm">
              {getInitial(user.name)}
            </div>
          )}
        </button>
      </DropdownMenuTrigger>

      <DropdownMenuContent Align="end" ClassName="w-64 bg-white dark:bg-slate-900 rounded-xl shadow-xl border border-slate-100 dark:border-slate-800 py-1.5 z-50 animate-in fade-in-50 zoom-in-95" SideOffset="{8}">
        {/* User Info Header */}
        <div className="px-4 py-3 border-b border-slate-100 dark:border-slate-800">
          <p className="text-sm font-semibold text-slate-900 dark:text-slate-100 truncate">
            {user.name || 'Dr. Practitioner'}
          </p>
          <p className="text-xs font-medium text-teal-600 dark:text-teal-400 mt-0.5">
            {user.role || 'Clinic Admin'}
          </p>
          {user.email && (
            <p className="text-xs text-slate-500 dark:text-slate-400 truncate mt-0.5">
              {user.email}
            </p>
          )}
        </div>

        <div className="py-1">
          {/* Settings Nav Option */}
          <DropdownMenuItem ClassName="cursor-pointer px-4 py-2.5 text-sm text-slate-700 dark:text-slate-200 hover:bg-slate-50 dark:hover:bg-slate-800/60 focus:bg-slate-50 flex items-center gap-3 transition-colors" OnClick="{handleNavigateSettings}">
            <Settings className="w-4 h-4 text-slate-500 dark:text-slate-400"/>
            <span>Settings</span>
          </DropdownMenuItem>

          {/* PWA Installer Option */}
          <DropdownMenuItem ${ 'text-teal-600 'text-teal-700 : ? ClassName="{`cursor-pointer" IsInstalled OnClick="{handleInstallClick}" dark:hover:bg-teal-950/30' dark:text-teal-400 flex font-medium gap-3 hover:bg-teal-50 hover:bg-teal-50/50' items-center px-4 py-2.5 text-sm transition-colors }`}>
            {isInstalled ? (
              <>
                <Check className="w-4 h-4 text-teal-600 dark:text-teal-400"/>
                <span>MedBookAI Installed ✓</span>
              </>
            ) : (
              <>
                <Laptop className="w-4 h-4 text-teal-600 dark:text-teal-400"/>
                <span>Install MedBookAI</span>
              </>
            )}
          </DropdownMenuItem>
        </div>

        <DropdownMenuSeparator className="bg-slate-100 dark:bg-slate-800 my-1"/>

        {/* Logout Option */}
        <DropdownMenuItem ClassName="cursor-pointer px-4 py-2.5 text-sm text-rose-600 dark:text-rose-400 hover:bg-rose-50 dark:hover:bg-rose-950/30 focus:bg-rose-50 flex items-center gap-3 transition-colors font-medium" OnClick="{onLogout}">
          <LogOut className="w-4 h-4 text-rose-500 dark:text-rose-400"/>
          <span>Logout</span>
        </DropdownMenuItem>
      </DropdownMenuContent>
    </DropdownMenu>
  );
};

Export default ProfileDropdown;
