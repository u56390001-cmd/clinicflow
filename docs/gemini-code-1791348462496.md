-- 1. Add-ons Master Catalog Table
CREATE TABLE public.addons (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    slug TEXT UNIQUE NOT NULL, -- e.g., 'whatsapp-ai-receptionist'
    name TEXT NOT NULL,
    category TEXT NOT NULL CHECK (category IN ('Automation', 'Growth & Marketing', 'Operations')),
    description TEXT NOT NULL,
    price_pkr NUMERIC(10,2) NOT NULL, -- Price in PKR / INR
    billing_period TEXT NOT NULL DEFAULT 'month',
    is_quantity_based BOOLEAN DEFAULT false,
    badge_text TEXT DEFAULT NULL,
    created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. Clinic Active Add-ons Subscriptions Table
CREATE TABLE public.clinic_addons (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    clinic_id UUID NOT NULL REFERENCES public.clinics(id) ON DELETE CASCADE,
    addon_id UUID NOT NULL REFERENCES public.addons(id) ON DELETE CASCADE,
    status TEXT NOT NULL CHECK (status IN ('active', 'pending_approval', 'cancelled', 'expired')) DEFAULT 'active',
    quantity INT NOT NULL DEFAULT 1,
    metadata JSONB DEFAULT '{}'::jsonb, -- e.g., { "ai_credits": 1000 }
    activated_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW(),
    UNIQUE(clinic_id, addon_id)
);

-- Enable RLS
ALTER TABLE public.addons ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.clinic_addons ENABLE ROW LEVEL SECURITY;

-- RLS Policies
CREATE POLICY "Everyone can read addons catalog" ON public.addons
    FOR SELECT USING (true);

CREATE POLICY "Users can view their clinic addons" ON public.clinic_addons
    FOR SELECT USING (
        clinic_id IN (
            SELECT clinic_id FROM public.clinic_members WHERE user_id = auth.uid()
        )
    );

CREATE POLICY "Clinic owners can manage their addons" ON public.clinic_addons
    FOR ALL USING (
        clinic_id IN (
            SELECT clinic_id FROM public.clinic_members 
            WHERE user_id = auth.uid() AND role IN ('owner', 'admin')
        )
    );

-- Helper Function for Feature Gatekeeping
CREATE OR REPLACE FUNCTION public.is_addon_active(p_clinic_id UUID, p_addon_slug TEXT)
RETURNS BOOLEAN AS $$
BEGIN
    RETURN EXISTS (
        SELECT 1 
        FROM public.clinic_addons ca
        JOIN public.addons a ON ca.addon_id = a.id
        WHERE ca.clinic_id = p_clinic_id 
          AND a.slug = p_addon_slug 
          AND ca.status = 'active'
    );
END;
$$ LANGUAGE plpgsql SECURITY DEFINER;
