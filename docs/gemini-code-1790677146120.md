export type HistoryCategory = 
  | 'past_illness' 
  | 'surgery' 
  | 'hospitalization' 
  | 'family_history' 
  | 'lifestyle' 
  | 'immunization';

export type HistorySource = 'patient_intake' | 'ai_ocr' | 'doctor_entry' | 'receptionist';
export type VerificationStatus = 'verified' | 'pending_approval';
export type ClinicalStatus = 'active' | 'resolved' | 'chronic';

export interface HistoryItem {
  id: string;
  patient_id: string;
  category: HistoryCategory;
  title: string;                  // e.g. "Diabetes Type 2", "Appendectomy"
  onset_date?: string;            // e.g. "2019" or "5 years ago"
  clinical_status: ClinicalStatus;
  source: HistorySource;
  verification_status: VerificationStatus;
  notes?: string;                 // Optional clinical note
  created_at: string;
  created_by_name?: string;
}

export interface LegacyHistoryFields {
  past_illnesses?: string;
  past_surgeries?: string;
  hospitalizations?: string;
  family_history?: string;
  personal_history?: string;
  immunization_history?: string;
}

export interface HistorySummaryStats {
  chronic_count: number;
  chronic_preview: string;       // e.g. "Diabetes T2, HTN"

  surgeries_preview: string;     // e.g. "Appendectomy (2018)"
  critical_risks_count: number;
  critical_risks_preview: string;// e.g. "Penicillin Allergy, Smoker"

  pending_ai_preview: string;    // e.g. "2 OCR items pending approval"
