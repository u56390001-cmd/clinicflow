CREATE TABLE IF NOT EXISTS public.clinic_invites (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  clinic_id UUID NOT NULL REFERENCES public.clinics(id) ON DELETE CASCADE,
  email VARCHAR(255) NOT NULL,
  role VARCHAR(50) NOT NULL CHECK (role IN ('owner', 'clinic_admin', 'doctor', 'receptionist', 'nurse', 'accountant', 'admin', 'staff')),
  permissions JSONB DEFAULT '{}'::jsonb,
  token VARCHAR(128) UNIQUE NOT NULL,
  invited_by UUID NOT NULL REFERENCES auth.users(id) ON DELETE CASCADE,
  expires_at TIMESTAMPTZ DEFAULT (NOW() + INTERVAL '7 days'),
  accepted_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enable RLS & Indexing
ALTER TABLE public.clinic_invites ENABLE ROW LEVEL SECURITY;

CREATE INDEX IF NOT EXISTS idx_clinic_invites_token 
ON public.clinic_invites(token) WHERE accepted_at IS NULL;

-- RLS Policy: Only Clinic Owners & Admins can read/create invites
CREATE POLICY "Tenant Isolation for Clinic Invites"
ON public.clinic_invites FOR ALL TO authenticated
USING (clinic_id IN (SELECT public.get_user_clinic_ids()))
WITH CHECK (clinic_id IN (SELECT public.get_user_clinic_ids()));


Task 2: Server Actions (lib/actions/onboarding.ts & lib/actions/invites.ts)
createClinicOnboardingAction(payload):

Validates input (name, organization_type, facility_size, phone, address, city, timezone).

Generates URL-safe unique slug from clinic name (e.g., skin-dots-clinic).

Inserts record into public.clinics.

Inserts logged-in user into public.clinic_members with role: 'owner'.

Writes CLINIC_CREATED event in public.audit_logs.

Returns redirect URL: /app/dashboard?clinic_id=${clinic.id}.

createStaffInviteAction(payload):

Enforces requirePermission(clinicId, 'staff:manage').

Generates a 64-character cryptographic token (crypto.randomBytes(32).toString('hex')).

Inserts row into public.clinic_invites.

Returns invite URL: ${NEXT_PUBLIC_SITE_URL}/invite/accept?token=${token}.

acceptInviteAction(token):

Validates token status (accepted_at IS NULL and expires_at > NOW()).

If authenticated: Inserts user into public.clinic_members with assigned role/permissions, marks token as accepted, and logs STAFF_INVITE_ACCEPTED in public.audit_logs.

If unauthenticated: Redirects to /signup?invite_token=${token} with pre-filled email.
